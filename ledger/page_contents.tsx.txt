      f.forEach((x: F) => {
        const k = key(x);
        if (!m[k]) m[k] = { k, n: 0, w: 0, l: 0, v: 0, o: 0, pend: 0, units: 0, staked: 0, dsum: 0 };
        const a = m[k];
        a.n += 1; a.dsum += x.d;
        if (x.r === "hit") a.w += 1; else if (x.r === "miss") a.l += 1; else if (x.r === "void") a.v += 1;
        else if (x.r === "pending") a.pend += 1; else a.o += 1;
        if ((x.r === "hit" || x.r === "miss") && x.p !== null) { a.units += x.p; a.staked += 1; }
      });
      return Object.values(m);
    };
    const order = (rows: Agg[], natural: (a: Agg, b: Agg) => number): Agg[] =>
      rows.slice().sort(rsort === "units" ? (a: Agg, b: Agg) => b.units - a.units
        : rsort === "rate" ? (a: Agg, b: Agg) => aggRate(b) - aggRate(a)
        : rsort === "slips" ? (a: Agg, b: Agg) => b.n - a.n : natural);
    const alpha = (last: string) => (a: Agg, b: Agg) => (a.k === last ? 1 : b.k === last ? -1 : a.k.localeCompare(b.k));
    const tgt = (a: Agg, b: Agg) => parseInt(a.k, 10) - parseInt(b.k, 10);
    return {
      total: f.length, onlyFlagged,
      book: order(by((x: F) => x.b, f), alpha("Mixed books")),
      target: order(by((x: F) => `+${x.t}`, f), tgt),
      sport: order(by((x: F) => x.sp, f), alpha("Unknown")),
      fbook: by((x: F) => x.b, ff), ftarget: by((x: F) => `+${x.t}`, ff), fsport: by((x: F) => x.sp, ff),
    };
  }, [rs, rl, rsort]);

  const userCells = (b: UB, label: string, withOdds: boolean): ReactNode[] => {
    const amt = [b.net === null ? "" : money(b.net), b.units === null ? "" : units(b.units)].filter((x: string) => x).join(" / ") || "-";
    const cells = [label, `${b.w}-${b.l}${b.p ? ` (${b.p} push)` : ""}`, "-", b.w + b.l ? `${((100 * b.w) / (b.w + b.l)).toFixed(1)}%` : "-", amt, "-"];
    return (withOdds ? [...cells, b.dn ? decToAm(b.dsum / b.dn) : "-"] : cells).map(B);
  };
  const flagCells = (a: Agg, label: string, withOdds: boolean): ReactNode[] => {
    const r = aggRow(a, withOdds);
    return [label, ...r.slice(1)].map(R);
  };
  const merged = (slips: Agg[], mine: UB[], withOdds: boolean, flg: Agg[] = [], flagMain: boolean = false): ReactNode[][] => {
    const rows: ReactNode[][] = [];
    const used = new Set<string>();
    const fused = new Set<string>();
    slips.forEach((a: Agg) => {
      rows.push(flagMain ? flagCells(a, a.k, withOdds) : aggRow(a, withOdds));
      const fa = flg.find((x: Agg) => x.k === a.k);
      if (fa) { fused.add(fa.k); rows.push(flagCells(fa, "\u21b3 \ud83d\udd34 Flagged", withOdds)); }
      const m = mine.find((b: UB) => b.k === a.k);
      if (m) { used.add(m.k); rows.push(userCells(m, "\u21b3 You", withOdds)); }
    });
    flg.filter((x: Agg) => !fused.has(x.k)).forEach((x: Agg) => rows.push(flagCells(x, `\ud83d\udd34 Flagged \u00b7 ${x.k}`, withOdds)));
    mine.filter((b: UB) => !used.has(b.k)).forEach((b: UB) => rows.push(userCells(b, `You \u00b7 ${b.k}`, withOdds)));
    return rows;
  };

  const u = DATA.user;
  const uw = u.reduce((a: number, r: UserRow) => a + r.wins, 0);
  const ul = u.reduce((a: number, r: UserRow) => a + r.losses, 0);
  const up = u.reduce((a: number, r: UserRow) => a + (r.pushes ?? 0), 0);
  const unet = u.filter((r: UserRow) => r.net !== null).reduce((a: number, r: UserRow) => a + (r.net ?? 0), 0);
  const uunits = u.filter((r: UserRow) => r.units !== null);
  const uu = uunits.reduce((a: number, r: UserRow) => a + (r.units ?? 0), 0);

  return (
    <Page title="Parlay Tracker" width="wide"
      summary={`Did the displayed parlay slips hit? Auto-graded from final box scores, plus your own reported results tracked separately. Updated ${DATA.updated}.`}>
      <Section heading="Displayed slips: overall">
        <Text>Legend: {R("red")} = flagged-price slips (a leg priced far off the other books; may be stale or a boost, check in app), {B("blue")} = your bets as you reported them, default color = regular slips (v1, v2 and v3). The three are shown side by side and never added together; the totals below are regular slips only.</Text>
        <Text>Each unique slip counts once per run at 1 unit, at the displayed odds. Hit rate is hits divided by settled slips.</Text>
        <Metrics metrics={[
          { value: `${o.hit}-${o.miss}`, label: "Hits-misses", detail: `${o.n} slips logged` },
          { value: rate(o), label: "Hit rate", detail: "settled slips only" },
          { value: units(o.profit), label: "Profit", detail: `${o.staked} units staked`, tone: o.profit >= 0 ? "success" : "danger" },
          { value: `${o.pending}`, label: "Pending", detail: o.other ? `${o.other} void or ungraded` : "games not final" },
        ]} />
        {DATA.flagged.n ? <Text>{R(`\ud83d\udd34 Flagged-price slips (${flagKeys.join(", ")}, tracked apart; off by default since v3): ${DATA.flagged.hit}-${DATA.flagged.miss} \u00b7 hit rate ${rate(DATA.flagged)} \u00b7 ${units(DATA.flagged.profit)} \u00b7 ${DATA.flagged.pending} pending \u00b7 ${DATA.flagged.n} slips logged`)}</Text> : null}
        <Text>{B(`Your bets: ${uw}-${ul}${up ? ` (${up} push)` : ""} \u00b7 win rate ${uw + ul ? `${((100 * uw) / (uw + ul)).toFixed(1)}%` : "-"} \u00b7 net ${money(unet)}${uunits.length ? ` / ${units(uu)}` : ""} \u00b7 ${u.length} days logged`)}</Text>
        {DATA.bankroll ? <Text>{B(`Bankroll: ${usd(DATA.bankroll.current)} (started ${usd(DATA.bankroll.start)} on ${DATA.bankroll.start_date}${DATA.bankroll.deposits ? `, deposits ${usd(DATA.bankroll.deposits)}` : ""}${DATA.bankroll.withdrawals ? `, withdrawals ${usd(DATA.bankroll.withdrawals)}` : ""}, your bets net ${money(DATA.bankroll.bets_net)} over ${DATA.bankroll.n_bets} bets${DATA.bankroll.n_no_net ? `; ${DATA.bankroll.n_no_net} without net $ not counted` : ""}) \u00b7 flat 0.5-1% stake now ${usd(DATA.bankroll.stake_min)}-${usd(DATA.bankroll.stake_max)}`)}</Text> : null}
      </Section>
      <Section heading="Breakdown">
        <Segmented<Dim> value={dim} onChange={setDim} options={[
          { value: "strategy", label: "Strategy" }, { value: "list", label: "List" },
          { value: "report", label: "Report" }, { value: "props", label: "Prop sides" }]} />
        <Table columns={["Group", "Slips", "Hit", "Miss", "Pending", "Hit rate", "Profit"]}
          rows={[...DATA.breakdown[dim].map((t: T & { k: string }): ReactNode[] => [t.k, `${t.n}`, `${t.hit}`, `${t.miss}`, `${t.pending}`, rate(t), units(t.profit)]),
            ...(DATA.fBreakdown[dim] ?? []).map((t: T & { k: string }): ReactNode[] => [`\ud83d\udd34 ${dim === "strategy" ? t.k : `Flagged \u00b7 ${t.k}`}`, `${t.n}`, `${t.hit}`, `${t.miss}`, `${t.pending}`, rate(t), units(t.profit)].map(R))]} />
        <Text>Viceroy's slips only. Red rows are the flagged-price slips (strategies ending in -flagged), kept apart from the regular totals. A slip shown in both reports counts in each. Records by book, odds and sport are in the next section. Strategy v1 is the original generator; v2 (from 2026-10-23) drops off-market prices, requires 3 books behind the fair price and a second sportsbook within 3 points, and shows only single-book slips of at most 2 legs. v3 (from 2026-10-29) keeps v2 and only allows a player-prop leg when 4+ books quote it and their fair prices sit within 4 points; it also price-shops every slip across books.{DATA.shop ? ` Price shop (v3): another book paid more for the same legs on ${DATA.shop.better} of ${DATA.shop.n} slips${DATA.shop.better ? `, by ${DATA.shop.avg_gain_pct}% on average` : ""}.` : ""}</Text>
      </Section>
      <Section heading="Records by sportsbook, odds and sport">
        <Segmented<string> value={rs} onChange={setRs} options={[{ value: "all", label: "All strategies" }, ...strategies.map((x: string) => ({ value: x, label: x })),
          ...(hasFlagged ? flagKeys.map((k: string) => ({ value: k, label: "\ud83d\udd34 " + k + " (red)" })) : [])]} />
        <Segmented<string> value={rl} onChange={setRl} options={[{ value: "all", label: has3leg ? "All lists" : "SGP + multi" }, { value: "sgp", label: "SGP" }, { value: "multi", label: "Multi-game" },
          ...(has3leg ? [{ value: "3leg", label: "3-leg" }] : [])]} />
        <Segmented<string> value={rsort} onChange={setRsort} options={[{ value: "natural", label: "Default order" }, { value: "units", label: "Sort by units" }, { value: "rate", label: "Sort by hit rate" }, { value: "slips", label: "Sort by slips" }]} />
        <Section heading="By sportsbook">
          <Table columns={["Book", "Record (W-L)", "Pending", "Hit rate", "Units / $", "ROI"]} rows={merged(recs.book, DATA.userBreak.book ?? [], false, recs.fbook, recs.onlyFlagged)} />
        </Section>
        <Section heading="By odds target">
          <Table columns={["Target", "Record (W-L)", "Pending", "Hit rate", "Units / $", "ROI", "Avg odds"]} rows={merged(recs.target, DATA.userBreak.odds ?? [], true, recs.ftarget, recs.onlyFlagged)} />
        </Section>
        <Section heading="By sport">
          <Table columns={["Sport", "Record (W-L)", "Pending", "Hit rate", "Units / $", "ROI"]} rows={merged(recs.sport, DATA.userBreak.sport ?? [], false, recs.fsport, recs.onlyFlagged)} />
        </Section>
        <Text>W is a slip that hit, L one that missed; 1 unit per unique slip per run at the displayed odds, and ROI is units over settled slips. Each slip counts once, at the target where it first appeared. Mixed books are v1 best-price slips split across books. Avg odds converts the average decimal odds at that target back to American. Red rows are the flagged-price slips (they follow the SGP/multi filter, appear under All strategies, and are the only rows when the red v2-flagged filter is chosen; never added into the regular rows). Blue rows are your bets: they ignore the strategy and SGP/multi filters, show net $ instead of units, and sit under the matching row (or at the bottom when nothing matches, including Untagged). Your odds buckets: under +150 (including minus odds) is +100, +150 to +249 is +200, +250 to +349 is +300, +350 and longer is +400.</Text>
      </Section>
      {DATA.clv.length ? (
        <Section heading="Closing line value (game lines only)">
          <Table columns={["Strategy", "Legs with a close", "Avg CLV", "Beat the close", "ML / Spread / Total avg"]}
            rows={DATA.clv.map((c: { k: string; n: number; avg: number; beat: number; mkt?: Record<string, [number, number]> }): ReactNode[] => {
              const pp = (x: number): string => `${x >= 0 ? "+" : ""}${x.toFixed(2)}`;
              const split = ["ml", "spread", "total"].map((m: string) => (c.mkt && c.mkt[m] ? `${pp(c.mkt[m][1])} (${c.mkt[m][0]})` : "-")).join(" / ");
              const cells = [c.k, `${c.n}`, `${pp(c.avg)} pp`, `${c.beat.toFixed(1)}%`, split];
              return isFlag(c.k) ? cells.map(R) : cells;
            })} />
          <Text>Close is the DraftKings line ESPN shows after the game, compared on the same line. Positive CLV means the price taken implied a lower probability than the close. The last column is the average CLV in pp by market (legs in brackets). Player props have no free closing source.</Text>
        </Section>
      ) : null}
      <Section heading="By run">
        <Table columns={["Run (UTC)", "Slips", "Hit", "Miss", "Pending", "Hit rate", "Profit"]}
          rows={DATA.byRun.map((t: T & { run: string; flagged?: boolean }): ReactNode[] => {
            const cells = [t.run, `${t.n}`, `${t.hit}`, `${t.miss}`, `${t.pending}`, rate(t), units(t.profit)];
            return t.flagged ? cells.map(R) : cells;
          })} />
      </Section>
      {DATA.userPays && DATA.userPays.nBets ? (
        <Section heading="Your reported bets: what pays">
          <Segmented<string> value={pd} onChange={setPd} options={[{ value: "sport", label: "Sport" }, { value: "legs", label: "Legs" },
            { value: "type", label: "Bet type" }, { value: "odds", label: "Odds range" }, { value: "book", label: "Sportsbook" }]} />
          <Table columns={["Group", "Bets", "W-L-P", "Staked", "Net $", "ROI"]}
            rows={[...(DATA.userPays.dims[pd] ?? []), { ...DATA.userPays.overall, k: "Overall" }].map((r: UPRow): ReactNode[] => [r.k, `${r.n}`, `${r.w}-${r.l}-${r.p}`,
              r.sn ? `$${r.staked.toFixed(2)}${r.sn < r.n ? ` (${r.sn} of ${r.n})` : ""}` : "-", r.netN ? money(r.net) : "-",
              r.sn && r.staked ? `${r.snet >= 0 ? "+" : ""}${((100 * r.snet) / r.staked).toFixed(1)}%${r.sn < r.n ? ` (${r.sn} bets)` : ""}` : "-"].map(B))} />
          <Text>{B(`Your bets only (never the graded slips). ROI is net over staked, counting only bets with a known stake. Missing: ${DATA.userPays.gaps.join("; ")}.`)}</Text>
          {DATA.userPays.roi ? <Text>{B(`ROI on logged stakes: ${pct(DATA.userPays.roi.logged)} \u00b7 including stakes derived from net $ and odds for ${DATA.userPays.roi.all.derived ?? 0} older bets: ${pct(DATA.userPays.roi.all)}.`)}</Text> : null}
          <Text>{B(`Small sample: ${DATA.userPays.nBets} bets in all. A group under 10 bets is not meaningful yet; one long-odds win swings its ROI.`)}</Text>
        </Section>
      ) : null}
      <Section heading="Your reported results">
        <Text>{B("Your own bets by day, as you reported them. Your totals are in the overall section and your records sit under each row of the tables above.")}</Text>
        <Table columns={["Date", "W-L-P", "Net $", "Units", "Note", "Entered (UTC)"]}
          rows={u.slice().reverse().map((r: UserRow) => [r.date, `${r.wins}-${r.losses}-${r.pushes ?? 0}`,
            r.net === null ? "-" : money(r.net), r.units === null ? "-" : units(r.units),
            r.note ?? "", r.entered_at.slice(0, 16).replace("T", " ")].map(B))} />
      </Section>
      <Section heading="How grading works">
        <BulletList items={[
          "Legs are settled from ESPN's public box scores. A confirmed DNP or postponed game is void; a stat that can't be found is ungraded, never guessed, until you confirm it and it is settled as a manual void (or other result) from a saved override.",
          "A slip is a miss if any leg lost. It's a hit if every remaining leg won. Void and push legs come out and the odds are recomputed without them.",
          "SGP odds are estimates; a book's real SGP price is usually shorter. Underdog pick'em pays from its own payout table.",
          ...DATA.lost.map((x: string) => `Not recoverable: ${x}.`),
        ]} />
      </Section>
    </Page>
  );
}
